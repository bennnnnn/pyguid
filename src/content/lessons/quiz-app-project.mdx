## Starter quiz

<PythonExample
  filename="quiz.py"
  code={`questions = [
    {
        "prompt": "What is 2 + 2?",
        "answer": "4"
    },
    {
        "prompt": "Capital of France?",
        "answer": "Paris"
    }
]

score = 0

# Simulated answers from the "player"

player_answers = ["4", "London", "Paris"]

for i, q in enumerate(questions):
given = player_answers[i]
print(q["prompt"])
print("You said:", given)
if given == q["answer"]:
print("Correct!")
score += 1
else:
print("Wrong. Answer:", q["answer"])
print()

print("Final score:", score, "/", len(questions))`}
/>
