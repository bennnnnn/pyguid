Combine **Flask**, **dict/list storage**, and **JSON** into a tiny **REST-style** API — the shape of real backends before databases.

## In-memory store

```python
from flask import Flask, jsonify, request

app = Flask(__name__)
todos = []
next_id = 1

@app.get("/todos")
def list_todos():
    return jsonify(todos)

@app.post("/todos")
def create_todo():
    global next_id
    data = request.get_json()
    item = {"id": next_id, "text": data["text"], "done": False}
    next_id += 1
    todos.append(item)
    return jsonify(item), 201

@app.patch("/todos/<int:todo_id>")
def mark_done(todo_id):
    for item in todos:
        if item["id"] == todo_id:
            item["done"] = True
            return jsonify(item)
    return jsonify({"error": "not found"}), 404
```
